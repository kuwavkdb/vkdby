# See https://www.robotstxt.org/robotstxt.html for documentation on how to use the robots.txt file

# Search results with a free-text query (q) on the people / units lists, and site search, have
# practically unlimited combinations, little value as search engine entries, and cause heavy queries
# when crawled (issue #1439, #1855). Tag / part / blood / hometown / status filter pages stay crawlable.
User-agent: *
Disallow: /people?*q=
Disallow: /units?*q=
Disallow: /search
# Combinations of 4 or more filter parameters are crawled exhaustively and multiply without limit (issue #1863).
# Filter pages with 1 to 3 parameters stay crawlable.
Disallow: /people?*&*&*&
Disallow: /units?*&*&*&

Sitemap: https://vkdb.jp/sitemap.xml
